function otsu(hist,total){let sum=0;for(let i=0;i<256;i++)sum+=i*hist[i];let n=0,s=0,best=-1,t=128;for(let i=0;i<256;i++){n+=hist[i];s+=i*hist[i];if(!n||n===total)continue;const v=n*(total-n)*(s/n-(sum-s)/(total-n))**2;if(v>best){best=v;t=i}}return t;}
function detectPaths(data,w,h,mode,adjust=0){
 const g=new Uint8Array(w*h),hist=new Uint32Array(256);for(let i=0;i<g.length;i++){const k=i*4;g[i]=Math.round(.299*data[k]+.587*data[k+1]+.114*data[k+2]);hist[g[i]]++;}

  const pairs=[];for(let i=0;i<active.length;i++){const p=active[i],last=p.pts.at(-1),prev=p.pts[Math.max(0,p.pts.length-5)],dx=x-last.x;if(dx>gap)continue;const slope=clamp((last.y-prev.y)/Math.max(1,last.x-prev.x),-5,5),pred=last.y+slope*dx;for(let j=0;j<runs.length;j++){const dist=Math.abs(runs[j].y-pred);if(dist<=maxJump+dx*1.5&&Math.abs(runs[j].y-last.y)<=maxJump+dx*5)pairs.push({i,j,cost:dist+dx*.3});}}

  for(let i=active.length-1;i>=0;i--)if(x-active[i].pts.at(-1).x>gap){paths.push(active[i].pts);active.splice(i,1)}
  for(let j=0;j<runs.length;j++)if(!usedR.has(j))active.push({pts:[runs[j]]});
 }
